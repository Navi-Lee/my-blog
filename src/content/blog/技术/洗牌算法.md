- 洗牌算法

把一个数组里的元素随机打乱顺序（Fisher-Yates 洗牌算法）

```js
const shuffle =(arr)=>{
  for(let i=arr.length-1;i>0;i--){
    const j =Math.floor((i+1)*Math.random())
    //math.random生成[0,1)区间随机数
    [arr[j],arr[i]]=[arr[i],arr[j]]
  }
  return arr
}
```

- 继承套餐
  - 原型继承
    ```js
    function Person(){
    this.name='Navi'
    }

    Person.prototype.eating = function(){
    console.log("eating")
    }
    //相当于定义了一个person类

    function Student(){}

    Student.prototype = new Person()
    //prototype也是对象，可以换成一个person实例

    const stu = new Student()

    console.log(stu.name) // "juege"
    stu.eating()         // "eating"
    ```

  - 借用构造函数继承
    
    ```js
    function Person(name){
    this.name=name
    this.hobbies=['eat','sleep']
    }
    function Students(name,number){
      Person.call(this,name)
    // 此时 this 指向当前新创建的 Student 实例(s1,s2)
    // Person.call(this, name)
    // 相当于让Person里面的this指向当前 Student 实例
    // 从而把Person中定义的属性添加到 Student 实例上
    this.number=number
    //students自己的this
    }
    const s1 = new Students('Tom', 1001)
    const s2 = new Students('Jerry', 1002)

    console.log(s1)
    console.log(s2)
    // {
    //   name: 'Tom',
    //   hobbies: ['eat', 'sleep'],
    //   number: 1001
    // }

    ```
